## 🛠️ The "Sentinel-HSTS" Pure-Code Prototype

### 1. The Core Detection Logic (JavaScript)

This script acts as the "Prophecy Engine." It tests if a domain is ready for HSTS by simulating the strict requirements of the preload list.

```javascript
// Sentinel-HSTS Core Logic: The "Preload Prophecy" Test
async function checkHSTSPreparedness(domain) {
    const results = {
        hasHTTPS: false,
        headerFound: false,
        maxAge: 0,
        includesSubdomains: false,
        preloadReady: false,
        breakageRisk: []
    };

    try {
        // 1. Check for HTTPS Enforcement
        const response = await fetch(`https://${domain}`, { method: 'HEAD' });
        results.hasHTTPS = true;

        // 2. Parse the Strict-Transport-Security Header
        const hstsHeader = response.headers.get("Strict-Transport-Security");
        
        if (hstsHeader) {
            results.headerFound = true;
            results.maxAge = parseInt(hstsHeader.match(/max-age=(\d+)/)?.[1] || 0);
            results.includesSubdomains = hstsHeader.includes("includeSubDomains");
            results.preloadReady = hstsHeader.includes("preload");
        }

        // 3. The "Breakage" Simulation
        // We probe common internal subdomains to see if they lack SSL
        const commonSubs = ['dev', 'test', 'api', 'internal'];
        for (let sub of commonSubs) {
            try {
                await fetch(`http://${sub}.${domain}`, { mode: 'no-cors' });
            } catch (e) {
                results.breakageRisk.push(`${sub}.${domain}`);
            }
        }

        return results;
    } catch (error) {
        return "Domain unreachable or blocking probes.";
    }
}
```
